cmake_minimum_required(VERSION 3.24)
project(cryptagent LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "This package supports native Windows with MSVC only")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded")
set(SODIUM_INCLUDE_DIR "" CACHE PATH "Directory containing sodium.h")
set(SODIUM_LIBRARY "" CACHE FILEPATH "Official x64 static libsodium.lib")
set(CANDIDATE_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/phase2/cpp/cryptagent.cpp" CACHE FILEPATH "Inspected candidate source")
option(CRYPTAGENT_ASAN "Instrument wrapper with MSVC AddressSanitizer" OFF)
if(NOT EXISTS "${SODIUM_INCLUDE_DIR}/sodium.h" OR NOT EXISTS "${SODIUM_LIBRARY}")
  message(FATAL_ERROR "Set SODIUM_INCLUDE_DIR and SODIUM_LIBRARY; see WINDOWS_SETUP.md")
endif()
add_executable(fixture_test "${CANDIDATE_SOURCE}" phase3/harness/fixture_test.cpp)
target_include_directories(fixture_test PRIVATE phase2/cpp "${SODIUM_INCLUDE_DIR}")
target_compile_definitions(fixture_test PRIVATE SODIUM_STATIC)
target_compile_options(fixture_test PRIVATE /W4 /permissive-)
target_link_libraries(fixture_test PRIVATE "${SODIUM_LIBRARY}")
if(CRYPTAGENT_ASAN)
  target_compile_options(fixture_test PRIVATE /Od /Zi /fsanitize=address)
  target_link_options(fixture_test PRIVATE /INCREMENTAL:NO)
endif()
enable_testing()
foreach(check roundtrip tamper aad fixed-nonce fixed-key empty boundary oversize-plaintext oversize-aad malformed wrong-key)
  add_test(NAME ${check} COMMAND fixture_test ${check} 17 16)
endforeach()
