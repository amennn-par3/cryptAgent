cmake_minimum_required(VERSION 3.20)
project(cryptagent_bfv_context LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
find_package(OpenFHE CONFIG REQUIRED)
set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} ${OpenFHE_CXX_FLAGS}")
include_directories(${OpenFHE_INCLUDE} ${OpenFHE_INCLUDE}/third-party/include ${OpenFHE_INCLUDE}/core ${OpenFHE_INCLUDE}/pke ${OpenFHE_INCLUDE}/binfhe)
link_directories(${OpenFHE_LIBDIR})
foreach(variant missing configured)
  add_executable(bfv_${variant}_pke main.cpp)
  target_compile_definitions(bfv_${variant}_pke PRIVATE CASE_NAME="bfv_${variant}_pke" CANDIDATE="../cases/bfv_${variant}_pke.hpp")
  target_link_libraries(bfv_${variant}_pke ${OpenFHE_SHARED_LIBRARIES})
endforeach()
