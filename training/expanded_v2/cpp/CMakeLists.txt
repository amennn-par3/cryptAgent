cmake_minimum_required(VERSION 3.20)
project(cryptagent_bfv_batch LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
find_package(OpenFHE CONFIG REQUIRED)
set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} ${OpenFHE_CXX_FLAGS}")
include_directories(${OpenFHE_INCLUDE} ${OpenFHE_INCLUDE}/third-party/include ${OpenFHE_INCLUDE}/core ${OpenFHE_INCLUDE}/pke ${OpenFHE_INCLUDE}/binfhe)
link_directories(${OpenFHE_LIBDIR})
foreach(variant any all)
  add_executable(bfv_batch_${variant} main.cpp)
  target_compile_definitions(bfv_batch_${variant} PRIVATE CASE_NAME="bfv_batch_${variant}" CANDIDATE="../cases/bfv_batch_${variant}.hpp")
  target_link_libraries(bfv_batch_${variant} ${OpenFHE_SHARED_LIBRARIES})
endforeach()
add_library(bfv_batch_delegate OBJECT delegate.cpp)
