cmake_minimum_required(VERSION 3.20)
project(cryptagent_heldout LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
find_package(OpenFHE CONFIG REQUIRED)
set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} ${OpenFHE_CXX_FLAGS}")
include_directories(${OpenFHE_INCLUDE} ${OpenFHE_INCLUDE}/third-party/include ${OpenFHE_INCLUDE}/core ${OpenFHE_INCLUDE}/pke ${OpenFHE_INCLUDE}/binfhe)
link_directories(${OpenFHE_LIBDIR})
foreach(variant leaky redacted unchecked bounded)
  if(variant STREQUAL "leaky" OR variant STREQUAL "redacted")
    set(kind telemetry)
  else()
    set(kind work)
  endif()
  add_executable(bfv_${kind}_${variant} main.cpp)
  target_compile_definitions(bfv_${kind}_${variant} PRIVATE CASE_NAME="bfv_${kind}_${variant}" CANDIDATE="../cases/bfv_${kind}_${variant}.hpp")
  if(kind STREQUAL "telemetry")
    target_compile_definitions(bfv_${kind}_${variant} PRIVATE TELEMETRY_CASE=1)
  endif()
  target_link_libraries(bfv_${kind}_${variant} ${OpenFHE_SHARED_LIBRARIES})
endforeach()
