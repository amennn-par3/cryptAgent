cmake_minimum_required(VERSION 3.20)
project(cryptagent_bfv_probes LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
find_package(OpenFHE CONFIG REQUIRED)
set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} ${OpenFHE_CXX_FLAGS}")
include_directories(${OpenFHE_INCLUDE} ${OpenFHE_INCLUDE}/third-party/include ${OpenFHE_INCLUDE}/core ${OpenFHE_INCLUDE}/pke ${OpenFHE_INCLUDE}/binfhe)
link_directories(${OpenFHE_LIBDIR})
foreach(case bfv_export_private bfv_export_public bfv_unbounded_sum bfv_bounded_sum bfv_accept_any bfv_validate_result)
  add_executable(${case} main.cpp)
  target_compile_definitions(${case} PRIVATE CASE_NAME="${case}" CANDIDATE="../cases/${case}.hpp")
  if(case MATCHES "export")
    target_compile_definitions(${case} PRIVATE KEY_PROBE)
  elseif(case MATCHES "sum")
    target_compile_definitions(${case} PRIVATE SUM_PROBE)
  endif()
  target_link_libraries(${case} ${OpenFHE_SHARED_LIBRARIES})
endforeach()
