cmake_minimum_required(VERSION 3.20)
project(cryptagent_repaired LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
find_package(OpenFHE CONFIG REQUIRED)
set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} ${OpenFHE_CXX_FLAGS}")
include_directories(${OpenFHE_INCLUDE} ${OpenFHE_INCLUDE}/third-party/include ${OpenFHE_INCLUDE}/core ${OpenFHE_INCLUDE}/pke ${OpenFHE_INCLUDE}/binfhe)
link_directories(${OpenFHE_LIBDIR})
foreach(variant correct arithmetic acceptance secret setup bounds)
  add_executable(bfv_${variant} main.cpp)
  target_compile_definitions(bfv_${variant} PRIVATE CASE_NAME="bfv_${variant}" CANDIDATE="../cases/bfv_${variant}.hpp")
  target_link_libraries(bfv_${variant} ${OpenFHE_SHARED_LIBRARIES})
endforeach()
