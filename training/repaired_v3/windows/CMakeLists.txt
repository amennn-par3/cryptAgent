cmake_minimum_required(VERSION 3.20)
project(cryptagent_cng_review LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "Windows CNG tests require native Windows; Linux results are not substitutes.")
endif()
set(CMAKE_CXX_STANDARD 20)
foreach(variant correct empty authentication publication keysize bounds)
  add_executable(aes_${variant} main.cpp)
  target_compile_definitions(aes_${variant} PRIVATE CASE_NAME="aes_${variant}" CANDIDATE="../cases/aes_${variant}.hpp")
  target_link_libraries(aes_${variant} bcrypt)
endforeach()
